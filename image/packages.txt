# Every Fedora package the WardOS host installs (the release image/Containerfile pins) (docs/desktop.md §Packages).
# One name per line, `# why` after it; blank lines and comment lines are ignored.
# The Containerfile installs from this file after enabling the COPRs of
# image/coprs.txt; desktop/install.sh layers it onto an existing Fedora;
# image/check-packages.sh (CI: "image packages") proves every name resolves in a
# fedora:<release> container with the same COPRs. The check, not this file, is the
# source of truth: a name that has not passed it yet is marked `# unverified` until it has
# (linux-firmware and hyprland-qtutils are unverified pending this change's packages job).
# Size matters (a first pull is the whole image): one browser, no compiler, the fonts
# the desktop renders, English locales; image/README.md "Size" says what each choice saves.

# --- runtime (ADR-0001: no developer toolchain on the host) -------------------------------
bubblewrap            # sandbox builder used by wardd (ADR-0002, ADR-0013)
git                   # snapshot/worktree plumbing (ADR-0010), `ward init` on repositories
curl                  # transport for `ward` self-checks and the egress relay tests
linux-firmware        # unverified # Wi-Fi/GPU/Bluetooth firmware; without it a laptop has no wireless (E-09: T480s had no Wi-Fi)
python3               # scripts/security-check and hook adapters
openssh-clients       # agent-side git-over-ssh through the credential broker
rustup                # rustup-init only, no toolchain: `wardos-install dev rust` puts one in ~/.rustup and ~/.cargo, which the verifier binds (image/README.md "Size")
podman                # nested containers inside project environments (ADR-0005)

# --- locales (image/README.md "Size": en_US and C.UTF-8; the rest are not installed) -------
glibc-langpack-en     # en_US.UTF-8 (the base image's glibc-all-langpacks, if present, is removed in the build)

# --- agents (ADR-0017: Claude Code, Codex and TamperWard ship in the image) ----------
nodejs24              # runtime for the three (Fedora ships versioned streams, no plain `nodejs`; Claude Code needs Node >= 22, which the build checks)
nodejs24-npm          # `npm ci` from image/agents/package-lock.json at build time; nothing installs at run time

# --- security posture (ADR-0017) ----------------------------------------------------------
firewalld             # host firewall, enabled, default zone `wardos` (nothing inbound); image/README.md "Security posture"

# --- compositor and session ---------------------------------------------------------
hyprland              # the compositor (ADR-0007) (COPR mineiro/hyprland)
hyprland-qtutils      # unverified # Hyprland's Qt dialogs (hyprland-dialog etc.), a runtime dep some prompts need (COPR mineiro/hyprland)
hyprlock              # lock screen (COPR mineiro/hyprland)
hypridle              # idle management (lock, screen off, suspend) (COPR mineiro/hyprland)
hyprpaper             # wallpaper daemon (image backgrounds; swaybg draws solid ones) (COPR mineiro/hyprland)
hyprpicker            # colour picker (wardos-capture color) (COPR mineiro/hyprland)
hyprsunset            # night light (wardos-toggle nightlight) (COPR mineiro/hyprland)
hyprpolkitagent       # polkit prompts on the desktop (pkexec) (COPR mineiro/hyprland)
xdg-desktop-portal-hyprland # screen sharing, screenshots for portals (COPR mineiro/hyprland)
xdg-desktop-portal-gtk # file chooser and settings portal
uwsm                  # Universal Wayland Session Manager: starts Hyprland as a systemd session (COPR mineiro/hyprland)
xorg-x11-server-Xwayland # X11 applications
qt6-qtwayland         # Qt 6 applications on Wayland
qt5-qtwayland         # Qt 5 applications on Wayland
polkit                # privilege prompts
gvfs                  # trash, MTP, network shares for nautilus
xdg-user-dirs         # ~/Documents, ~/Downloads, ...
xdg-utils             # xdg-open, xdg-mime, default applications
mesa-dri-drivers      # OpenGL for the compositor
mesa-vulkan-drivers   # Vulkan (Hyprland's renderer can use it; browsers do)

# --- shell surfaces until E-10 (rendered from ward-shell's models) --------------------
waybar                # the trust bar
fuzzel                # command centre and every menu
mako                  # notifications and approvals
libnotify             # notify-send (wardos-notify)
swayosd               # on-screen volume and brightness (COPR erikreider/swayosd)
