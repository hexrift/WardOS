# Every Fedora 42 package the WardOS host installs (docs/desktop.md §Packages).
# One name per line, `# why` after it; blank lines and comment lines are ignored.
# The Containerfile installs from this file after enabling the COPRs of
# image/coprs.txt; desktop/install.sh layers it onto an existing Fedora;
# image/check-packages.sh (CI: "image packages") proves every name resolves in a
# fedora:42 container with the same COPRs. `# unverified` marks a name that check has
# not confirmed yet; the check, not this file, is the source of truth.

# --- runtime (ADR-0001: no developer toolchain on the host, one explicit exception) ----
bubblewrap            # sandbox builder used by wardd (ADR-0002, ADR-0013)
git                   # snapshot/worktree plumbing (ADR-0010), `ward init` on repositories
curl                  # transport for `ward` self-checks and the egress relay tests
python3               # scripts/security-check and hook adapters
openssh-clients       # agent-side git-over-ssh through the credential broker
cargo                 # the trusted verifier binds a host toolchain read-only into Zone 2
rust                  # (image/README.md: the one, temporary, toolchain exception)
podman                # nested containers inside project environments (ADR-0005)

# --- compositor and session ---------------------------------------------------------

hypridle              # idle management (lock, screen off, suspend) (COPR solopasha/hyprland) # unverified
hyprpaper             # wallpaper daemon (image backgrounds; swaybg draws solid ones) (COPR solopasha/hyprland) # unverified
hyprpicker            # colour picker (wardos-capture color) (COPR solopasha/hyprland) # unverified
hyprsunset            # night light (wardos-toggle nightlight) (COPR solopasha/hyprland) # unverified
hyprpolkitagent       # polkit prompts on the desktop (pkexec) (COPR solopasha/hyprland) # unverified
xdg-desktop-portal-hyprland # screen sharing, screenshots for portals

uwsm                  # Universal Wayland Session Manager: starts Hyprland as a systemd session (COPR solopasha/hyprland) # unverified

qt6-qtwayland         # Qt 6 applications on Wayland
qt5-qtwayland         # Qt 5 applications on Wayland

gvfs                  # trash, MTP, network shares for nautilus
xdg-user-dirs         # ~/Documents, ~/Downloads, ...

mesa-dri-drivers      # OpenGL for the compositor
